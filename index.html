  <body>

    <h2>SAB Voice Note Recorder</h2>


    <p><b>Episode:</b> <span id="episodeText"></span></p>


    <button onclick="startRecording()">🎤 Start Recording</button>

    <button onclick="stopRecording()">⏹ Stop Recording</button>

    <button onclick="uploadRecording()">⬆ Upload Note</button>


    <audio id="audioPlayback" controls></audio>


    <div id="status">Ready.</div>


    <form id="uploadForm" method="POST" target="hiddenFrame">

      <input type="hidden" name="episode" id="episodeInput">

      <input type="hidden" name="row" id="rowInput">

      <input type="hidden" name="audio" id="audioInput">

    </form>


    <iframe name="hiddenFrame" id="hiddenFrame"></iframe>


    <script>

      let mediaRecorder;

      let audioChunks = [];

      let audioBlob;


      const params = new URLSearchParams(window.location.search);

      const row = params.get("row");

      const episode = params.get("episode");

      const apiUrl = params.get("api");


      document.getElementById("episodeText").innerText = episode || "Not found";

      document.getElementById("episodeInput").value = episode || "";

      document.getElementById("rowInput").value = row || "";


      if (!row || !episode || !apiUrl) {

        document.getElementById("status").innerText =

          "Missing row, episode, or API URL. Please open this from the Google Sheet.";

      } else {

        document.getElementById("uploadForm").action = apiUrl;

      }


      async function startRecording() {

        try {

          document.getElementById("status").innerText =

            "Requesting microphone permission...";


          const stream = await navigator.mediaDevices.getUserMedia({

            audio: true

          });


          audioChunks = [];


          mediaRecorder = new MediaRecorder(stream);


          mediaRecorder.ondataavailable = function(event) {

            audioChunks.push(event.data);

          };


          mediaRecorder.onstop = function() {

            audioBlob = new Blob(audioChunks, {

              type: "audio/webm"

            });


            const audioUrl = URL.createObjectURL(audioBlob);

            document.getElementById("audioPlayback").src = audioUrl;


            document.getElementById("status").innerText =

              "Recording stopped. Please preview and upload.";

          };


          mediaRecorder.start();


          document.getElementById("status").innerText = "Recording...";

        } catch (error) {

          document.getElementById("status").innerText =

            "ERROR: " + error.name + " - " + error.message;

        }

      }


      function stopRecording() {

        if (mediaRecorder && mediaRecorder.state !== "inactive") {

          mediaRecorder.stop();

        }

      }


      function uploadRecording() {

        if (!audioBlob) {

          document.getElementById("status").innerText =

            "Please record something first.";

          return;

        }


        document.getElementById("status").innerText = "Preparing upload...";


        const reader = new FileReader();


        reader.onloadend = function() {

          document.getElementById("audioInput").value = reader.result;


          document.getElementById("status").innerText =

            "Uploading... Please wait.";


          document.getElementById("uploadForm").submit();


          setTimeout(function() {

            document.getElementById("status").innerText =

              "Upload submitted. Check Column L in the SAB sheet. It should change to 🎤 Play Note.";

          }, 3000);

        };


        reader.readAsDataURL(audioBlob);

      }

    </script>
